create or replace function rpc_admin_upsert_menu( p_token uuid, p_id text, p_name text, p_price_employee integer, p_price_executive integer, p_price_dispatch integer default 0, p_fax_number text default '', p_deadline_time text default '', p_deadline_time_show boolean default false, p_course_name text default '', p_course_name_show boolean default false, p_customer_number text default '', p_customer_number_show boolean default false ) returns table( id text, name text, price_employee integer, price_executive integer, price_dispatch integer, fax_number text, deadline_time text, deadline_time_show boolean, course_name text, course_name_show boolean, customer_number text, customer_number_show boolean ) language plpgsql security definer set search_path = public as $$ #variable_conflict use_column declare v_menu menu; begin perform _require_admin(p_token); if p_id is null or p_id = '' then insert into menu (name, price_employee, price_executive, price_dispatch) values (p_name, coalesce(p_price_employee,0), coalesce(p_price_executive,0), coalesce(p_price_dispatch,0)) returning * into v_menu; else update menu set name = p_name, price_employee = coalesce(p_price_employee,0), price_executive = coalesce(p_price_executive,0), price_dispatch = coalesce(p_price_dispatch,0) where menu.id = p_id returning * into v_menu; end if; insert into menu_fax ( menu_id, fax_number, deadline_time, deadline_time_show, course_name, course_name_show, customer_number, customer_number_show ) values ( v_menu.id, coalesce(p_fax_number,''), coalesce(p_deadline_time,''), coalesce(p_deadline_time_show,false), coalesce(p_course_name,''), coalesce(p_course_name_show,false), coalesce(p_customer_number,''), coalesce(p_customer_number_show,false) ) on conflict (menu_id) do update set fax_number = coalesce(p_fax_number,''), deadline_time = coalesce(p_deadline_time,''), deadline_time_show = coalesce(p_deadline_time_show,false), course_name = coalesce(p_course_name,''), course_name_show = coalesce(p_course_name_show,false), customer_number = coalesce(p_customer_number,''), customer_number_show = coalesce(p_customer_number_show,false); return query select v_menu.id, v_menu.name, v_menu.price_employee, v_menu.price_executive, v_menu.price_dispatch, coalesce(p_fax_number,''), coalesce(p_deadline_time,''), coalesce(p_deadline_time_show,false), coalesce(p_course_name,''), coalesce(p_course_name_show,false), coalesce(p_customer_number,''), coalesce(p_customer_number_show,false); end; $$; grant execute on function rpc_admin_upsert_menu(uuid, text, text, integer, integer, integer, text, text, boolean, text, boolean, text, boolean) to anon; -- ------------------------------------------------------------ -- 3. 管理タブのメニュー一覧: 新しい項目もまとめて返す -- （戻り値の列が変わるため一旦DROP） -- ------------------------------------------------------------ drop function if exists rpc_admin_list_menu(uuid); create or replace function rpc_admin_list_menu(p_token uuid) returns table( id text, name text, price_employee integer, price_executive integer, price_dispatch integer, fax_number text, deadline_time text, deadline_time_show boolean, course_name text, course_name_show boolean, customer_number text, customer_number_show boolean ) language plpgsql security definer set search_path = public as $$ #variable_conflict use_column begin perform _require_admin(p_token); return query select m.id, m.name, m.price_employee, m.price_executive, m.price_dispatch, coalesce(mf.fax_number, ''), coalesce(mf.deadline_time, ''), coalesce(mf.deadline_time_show, false), coalesce(mf.course_name, ''), coalesce(mf.course_name_show, false), coalesce(mf.customer_number, ''), coalesce(mf.customer_number_show, false) from menu m left join menu_fax mf on mf.menu_id = m.id order by m.name; end; $$; grant execute on function rpc_admin_list_menu(uuid) to anon;
